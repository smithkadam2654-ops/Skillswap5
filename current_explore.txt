import { mockUsers } from "../data/mockUsers";
import React, { useState, useEffect, useMemo, useCallback, useRef } from 'react';
import { Link, useNavigate } from 'react-router-dom';
import { motion, AnimatePresence } from 'motion/react';
import { toast, Toaster } from 'react-hot-toast';
import { 
  Search, Home, Compass, LayoutGrid, MessageSquare, 
  Settings, LogOut, Filter, CheckCircle2, Star, MapPin,
  Menu, X, Users, BookOpen, Repeat, ChevronDown, SlidersHorizontal, ChevronLeft
} from 'lucide-react';
import Navbar from '../components/Navbar';

// --- Types ---

// --- Dummy Data ---
const ALL_CATEGORIES = [
  'Programming', 'Design', 'Music', 'Languages', 'Photography', 
  'Finance', 'AI', 'Marketing', 'Sports'
];


// --- Custom Hooks ---
function useDebounce(value, delay) {
  const [debouncedValue, setDebouncedValue] = useState(value);
  useEffect(() => {
    const handler = setTimeout(() => {
      setDebouncedValue(value);
    }, delay);
    return () => clearTimeout(handler);
  }, [value, delay]);
  return debouncedValue;
}

// --- Main Component ---
export default function ExplorePage() {
  const navigate = useNavigate();
  const scrollRef = useRef(null);
  
  // -- UI State --
  const [isSidebarOpen, setIsSidebarOpen] = useState(false);
  const [isFilterOpen, setIsFilterOpen] = useState(false);
  const [isLogoutModalOpen, setIsLogoutModalOpen] = useState(false);
  const [isProfileModalOpen, setIsProfileModalOpen] = useState(false);
  const [isSkillsModalOpen, setIsSkillsModalOpen] = useState(false);
  const [selectedProfile, setSelectedProfile] = useState(null);
  const [isLoading, setIsLoading] = useState(false);

  const getInitialState = (key, defaultValue) => {
    const saved = sessionStorage.getItem('exploreState');
    if (saved) {
      const parsed = JSON.parse(saved);
      return parsed[key] !== undefined ? parsed[key] : defaultValue;
    }
    return defaultValue;
  };

  // -- Filter & Sort State --
  const [searchQuery, setSearchQuery] = useState(() => getInitialState('searchQuery', ''));
  const debouncedSearchQuery = useDebounce(searchQuery, 300);
  
  const [activeCategory, setActiveCategory] = useState(() => getInitialState('activeCategory', null));
  const [selectedLevels, setSelectedLevels] = useState(() => getInitialState('selectedLevels', []));
  const [selectedMode, setSelectedMode] = useState(() => getInitialState('selectedMode', null));
  const [selectedAvailability, setSelectedAvailability] = useState(() => getInitialState('selectedAvailability', []));
  const [minRating, setMinRating] = useState(() => getInitialState('minRating', 0));
  const [sortBy, setSortBy] = useState(() => getInitialState('sortBy', 'Newest'));

  // Save to session storage whenever these change
  useEffect(() => {
    const state = {
      searchQuery,
      activeCategory,
      selectedLevels,
      selectedMode,
      selectedAvailability,
      minRating,
      sortBy,
    };
    sessionStorage.setItem('exploreState', JSON.stringify(state));
  }, [searchQuery, activeCategory, selectedLevels, selectedMode, selectedAvailability, minRating, sortBy]);

  const [pendingLevels, setPendingLevels] = useState(selectedLevels);
  const [pendingMode, setPendingMode] = useState(selectedMode);
  const [pendingAvailability, setPendingAvailability] = useState(selectedAvailability);
  const [pendingMinRating, setPendingMinRating] = useState(minRating);

  // Sync pending state with active state when drawer opens
  useEffect(() => {
    if (isFilterOpen) {
      setPendingLevels(selectedLevels);
      setPendingMode(selectedMode);
      setPendingAvailability(selectedAvailability);
      setPendingMinRating(minRating);
    }
  }, [isFilterOpen, selectedLevels, selectedMode, selectedAvailability, minRating]);

  // Simulate loading on filter changes
  useEffect(() => {
    setIsLoading(true);
    const timer = setTimeout(() => setIsLoading(false), 400);
    return () => clearTimeout(timer);
  }, [debouncedSearchQuery, activeCategory, selectedLevels, selectedMode, selectedAvailability, minRating, sortBy]);

  // -- Handlers --
  const handleLogout = () => {
    setIsLogoutModalOpen(false);
    toast.success('Logged out successfully');
    navigate('/');
  };

  const handleApplyFilters = () => {
    setSelectedLevels(pendingLevels);
    setSelectedMode(pendingMode);
    setSelectedAvailability(pendingAvailability);
    setMinRating(pendingMinRating);
    setIsFilterOpen(false);
    toast.success('Filters applied');
  };

  const handleClearFilters = () => {
    setSearchQuery('');
    setActiveCategory(null);
    setSelectedLevels([]);
    setSelectedMode(null);
    setSelectedAvailability([]);
    setMinRating(0);
    setSortBy('Newest');
    
    setPendingLevels([]);
    setPendingMode(null);
    setPendingAvailability([]);
    setPendingMinRating(0);
    toast.success('Filters cleared');
  };

  const togglePendingLevel = (level) => {
    setPendingLevels(prev => prev.includes(level) ? prev.filter(l => l !== level) : [...prev, level]);
  };
  const togglePendingAvailability = (avail) => {
    setPendingAvailability(prev => prev.includes(avail) ? prev.filter(a => a !== avail) : [...prev, avail]);
  };
  const toggleCategory = (cat) => {
    setActiveCategory(prev => {
      const newCat = prev === cat ? null : cat;
      if (newCat) toast.success(`Category: ${newCat}`);
      else toast.success('Category filter removed');
      return newCat;
    });
  };

  const handleCardClick = (card) => {
    setIsLoading(true);
    setTimeout(() => {
      setIsLoading(false);
      setSelectedProfile(card);
      setIsProfileModalOpen(true);
      toast.success(`Profile opened: ${card.name}`);
    }, 500);
  };

  const handleSkillClick = (e, skill, type) => {
    e.stopPropagation();
    setSearchQuery(skill);
    toast.success(`Filtered by ${type}: ${skill}`);
  };

  const handleLocationClick = (e, location) => {
    e.stopPropagation();
    setSearchQuery(location);
    toast.success(`Filtered by location: ${location}`);
  };
  
  const handleRatingClick = (e) => {
    e.stopPropagation();
    setSortBy('Highest Rated');
    toast.success('Sorted by Highest Rated');
  };
  
  const handleSwapsClick = (e) => {
    e.stopPropagation();
    setSortBy('Most Swaps');
    toast.success('Sorted by Most Swaps');
  };

  const handleStatClick = (type) => {
    if (type === 'Mentors') {
      toast.success('Showing all mentors');
    } else if (type === 'Skills') {
      setIsSkillsModalOpen(true);
    } else if (type === 'Swaps') {
      setSortBy('Most Swaps');
      toast.success('Sorted by Most Swaps');
    } else if (type === 'Rating') {
      setSortBy('Highest Rated');
      toast.success('Sorted by Highest Rated');
    }
  };

  // -- Filtering & Sorting Logic --
  const [skillCards, setSkillCards] = useState(() => {
    const saved = localStorage.getItem('addedSkills');
    if (saved) {
      try {
        const parsed = JSON.parse(saved);
        return [...parsed, ...mockUsers];
      } catch (e) {
        return mockUsers;
      }
    }
    return mockUsers;
  });

  const filteredAndSortedCards = useMemo(() => {
    let result = [...skillCards];

    // Search
    if (debouncedSearchQuery) {
      const q = debouncedSearchQuery.toLowerCase();
      result = result.filter(card => 
        card.name.toLowerCase().includes(q) ||
        card.location.toLowerCase().includes(q) ||
        (card.college && card.college.toLowerCase().includes(q)) ||
        (card.branch && card.branch.toLowerCase().includes(q)) ||
        (card.profession && card.profession.toLowerCase().includes(q)) ||
        card.teaching.some(s => s.toLowerCase().includes(q)) ||
        card.wants.some(s => s.toLowerCase().includes(q))
      );
    }

    // Category
    if (activeCategory) {
      result = result.filter(card => card.category === activeCategory);
    }

    // Level
    if (selectedLevels.length > 0) {
      result = result.filter(card => selectedLevels.includes(card.level));
    }

    // Mode
    if (selectedMode) {
      result = result.filter(card => card.mode === selectedMode);
    }

    // Availability
    if (selectedAvailability.length > 0) {
      result = result.filter(card => 
        selectedAvailability.some(avail => card.availability.includes(avail))
      );
    }

    // Min Rating
    if (minRating > 0) {
      result = result.filter(card => card.rating >= minRating);
    }

    // Sort
    switch (sortBy) {
      case 'Newest':
        result.sort((a, b) => b.dateAdded - a.dateAdded);
        break;
      case 'Highest Rated':
        result.sort((a, b) => b.rating - a.rating);
        break;
      case 'Most Swaps':
        result.sort((a, b) => b.swaps - a.swaps);
        break;
      case 'Alphabetical':
        result.sort((a, b) => a.name.localeCompare(b.name));
        break;
      case 'Most Skills':
        result.sort((a, b) => (b.teaching.length + b.wants.length) - (a.teaching.length + a.wants.length));
        break;
      case 'Least Skills':
        result.sort((a, b) => (a.teaching.length + a.wants.length) - (b.teaching.length + b.wants.length));
        break;
    }

    return result;
  }, [skillCards, debouncedSearchQuery, activeCategory, selectedLevels, selectedMode, selectedAvailability, minRating, sortBy]);

  const allAvailableSkills = useMemo(() => {
    const skills = new Set();
    skillCards.forEach(c => {
      c.teaching.forEach(s => skills.add(s));
      c.wants.forEach(s => skills.add(s));
    });
    return Array.from(skills).sort();
  }, [skillCards]);

  // -- Render Helpers --
  const stats = [
    { label: 'Mentors', value: '1,250', icon: Users, color: 'text-blue-600', bg: 'bg-blue-50', onClick: () => handleStatClick('Mentors') },
    { label: 'Skills', value: '3,450', icon: BookOpen, color: 'text-purple-600', bg: 'bg-purple-50', onClick: () => handleStatClick('Skills') },
    { label: 'Swaps', value: '18,000', icon: Repeat, color: 'text-green-600', bg: 'bg-green-50', onClick: () => handleStatClick('Swaps') },
    { label: 'Avg Rating', value: '4.9', icon: Star, color: 'text-yellow-500', bg: 'bg-yellow-50', onClick: () => handleStatClick('Rating') },
  ];

  return (
    <motion.div 
      initial={{ opacity: 0, x: -20 }}
      animate={{ opacity: 1, x: 0 }}
      exit={{ opacity: 0, x: 20 }}
      transition={{ duration: 0.3 }}
      className="flex flex-col h-screen bg-[#F8FAFC] font-sans overflow-hidden text-[#111827]"
    >
      <Navbar />
      <Toaster position="bottom-right" />
      
      {/* Main Content */}
      <main className="flex-1 flex flex-col h-full overflow-hidden relative">
        <header className="lg:hidden flex items-center justify-between p-4 bg-white border-b border-slate-200 z-30">
          <span className="font-bold text-lg">SkillSwap</span>
          <button onClick={() => setIsFilterOpen(true)} className="p-2 -mr-2 text-slate-600"><Filter className="w-6 h-6" /></button>
        </header>

        <div ref={scrollRef} className="flex-1 overflow-y-auto p-4 md:p-6 lg:p-8 space-y-8 pb-24 lg:pb-8 hide-scrollbar">
          {/* Page Header */}
          <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} className="space-y-4">
            <Link to="/dashboard" className="inline-flex items-center gap-2 text-[#6B7280] hover:text-[#5B5BFF] transition-colors group text-[14px] font-medium mb-2">
              <ChevronLeft className="w-4 h-4 group-hover:-translate-x-1 transition-transform" />
              Back to Home
            </Link>
            <div className="flex items-center justify-between">
              <div>
                <h1 className="text-[32px] font-bold leading-tight">Explore Skills</h1>
                <p className="text-[15px] text-[#6B7280]">Find people who can teach you or learn with you.</p>
              </div>
            </div>
            <div className="relative max-w-2xl mt-6">
              <Search className="absolute left-4 top-1/2 -translate-y-1/2 text-slate-400 w-5 h-5" />
              <input 
                type="text"
                value={searchQuery}
                onChange={(e) => setSearchQuery(e.target.value)}
                placeholder="Search skills, technologies, hobbies..."
                className="w-full h-14 pl-12 pr-6 rounded-full bg-white border border-slate-200 shadow-sm focus:outline-none focus:border-[#5B5BFF] focus:ring-2 focus:ring-[#5B5BFF]/20 text-[15px] transition-all"
              />
            </div>
          </motion.div>

          {/* Categories */}
          <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} transition={{ delay: 0.1 }} className="flex overflow-x-auto gap-3 pb-2 hide-scrollbar">
            {ALL_CATEGORIES.map(cat => (
              <button 
                key={cat}
                onClick={() => toggleCategory(cat)}
                className={`h-11 px-5 rounded-full text-[15px] font-medium whitespace-nowrap transition-all duration-300 ${activeCategory === cat ? 'bg-gradient-to-r from-[#5B5BFF] to-[#7C3AED] text-white shadow-md shadow-[#5B5BFF]/30' : 'bg-white text-[#6B7280] border border-slate-200 hover:border-[#5B5BFF] hover:text-[#5B5BFF] hover:shadow-sm'}`}
              >
                {cat}
              </button>
            ))}
          </motion.div>

          {/* Stats */}
          <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
            {stats.map((stat, i) => (
              <motion.button 
                key={stat.label} 
                onClick={stat.onClick}
                initial={{ opacity: 0, y: 20 }} 
                animate={{ opacity: 1, y: 0 }} 
                transition={{ delay: i * 0.1 }} 
                className="bg-white rounded-2xl p-4 shadow-sm border border-slate-100 flex items-center gap-4 hover:-translate-y-1 hover:shadow-md transition-all text-left"
              >
                <div className={`w-12 h-12 rounded-xl flex items-center justify-center ${stat.bg} ${stat.color}`}>
                  <stat.icon className="w-6 h-6" />
                </div>
                <div>
                  <div className="text-[18px] font-bold">{stat.value}</div>
                  <div className="text-[13px] text-[#6B7280]">{stat.label}</div>
                </div>
              </motion.button>
            ))}
          </div>

          <div className="flex items-center justify-between">
            <h2 className="text-[22px] font-semibold">Results ({filteredAndSortedCards.length})</h2>
            
            {/* Sort Dropdown (Simple implementation) */}
            <div className="relative group">
              <button className="flex items-center gap-2 px-4 py-2 bg-white border border-slate-200 rounded-xl text-[14px] font-medium text-[#6B7280] hover:text-[#111827] transition-colors">
                <SlidersHorizontal className="w-4 h-4" />
                Sort: {sortBy}
                <ChevronDown className="w-4 h-4" />
              </button>
              <div className="absolute right-0 top-full mt-2 w-48 bg-white rounded-xl shadow-xl border border-slate-100 py-2 opacity-0 invisible group-hover:opacity-100 group-hover:visible transition-all z-20">
                {['Newest', 'Highest Rated', 'Most Swaps', 'Alphabetical', 'Most Skills', 'Least Skills'].map(s => (
                  <button key={s} onClick={() => setSortBy(s)} className={`w-full text-left px-4 py-2 text-[14px] hover:bg-slate-50 transition-colors ${sortBy === s ? 'text-[#5B5BFF] font-medium' : 'text-[#6B7280]'}`}>
                    {s}
                  </button>
                ))}
              </div>
            </div>
          </div>

          {/* Skill Cards */}
          <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-6">
            {isLoading ? (
              // Skeleton Loaders
              [...Array(6)].map((_, i) => (
                <div key={i} className="bg-white rounded-[20px] p-5 shadow-sm border border-slate-100 flex flex-col animate-pulse">
                  <div className="flex items-center gap-4 mb-5">
                    <div className="w-14 h-14 rounded-full bg-slate-200"></div>
                    <div className="flex-1 space-y-2">
                      <div className="h-5 bg-slate-200 rounded w-1/2"></div>
                      <div className="h-4 bg-slate-200 rounded w-3/4"></div>
                    </div>
                  </div>
                  <div className="flex-1 space-y-4 mb-6">
                    <div className="space-y-2">
                      <div className="h-4 bg-slate-200 rounded w-1/4"></div>
                      <div className="flex gap-2"><div className="h-6 bg-slate-200 rounded w-1/3"></div><div className="h-6 bg-slate-200 rounded w-1/4"></div></div>
                    </div>
                    <div className="space-y-2">
                      <div className="h-4 bg-slate-200 rounded w-1/4"></div>
                      <div className="flex gap-2"><div className="h-6 bg-slate-200 rounded w-1/3"></div></div>
                    </div>
                  </div>
                  <div className="pt-4 border-t border-slate-100 flex justify-between">
                    <div className="h-5 bg-slate-200 rounded w-1/4"></div>
                    <div className="h-10 bg-slate-200 rounded-xl w-1/3"></div>
                  </div>
                </div>
              ))
            ) : filteredAndSortedCards.length > 0 ? (
              filteredAndSortedCards.map((card, i) => (
                <motion.div 
                  key={card.id} 
                  initial={{ opacity: 0, scale: 0.95 }} 
                  animate={{ opacity: 1, scale: 1 }} 
                  transition={{ delay: 0.1 + (i * 0.05) }} 
                  className="bg-white rounded-[20px] p-5 shadow-sm hover:shadow-xl hover:shadow-[#5B5BFF]/10 border border-slate-100 hover:-translate-y-1 transition-all duration-300 flex flex-col"
                >
                  <div className="flex items-center gap-4 mb-5">
                    <button onClick={() => navigate(`/profile/${card.id}`)} className="relative group shrink-0">
                      <img src={card.avatar} alt={card.name} className="w-14 h-14 rounded-full object-cover border-2 border-slate-50 transition-transform group-hover:scale-105" />
                      {card.status === 'Online' && (
                        <span className="absolute bottom-0 right-0 w-3.5 h-3.5 bg-green-500 border-2 border-white rounded-full"></span>
                      )}
                    </button>
                    <div className="flex-1 min-w-0">
                      <div className="flex items-center justify-between">
                        <h3 className="text-[18px] font-bold cursor-pointer hover:text-[#5B5BFF] transition-colors truncate" onClick={() => navigate(`/profile/${card.id}`)}>{card.name}</h3>
                        <button className="flex items-center gap-1 hover:text-[#F59E0B] transition-colors shrink-0" onClick={handleRatingClick}>
                          <Star className="w-3.5 h-3.5 text-[#F59E0B] fill-current" /> <span className="font-medium text-sm">{card.rating}</span>
                        </button>
                      </div>
                      <p className="text-[13px] text-slate-500 font-medium truncate mt-0.5">{card.branch} • {card.year}</p>
                      <div className="flex items-center gap-1.5 text-[13px] text-[#6B7280] mt-1">
                        <button className="flex items-center gap-1 hover:text-[#5B5BFF] transition-colors truncate" onClick={(e) => handleLocationClick(e, card.location)}>
                          <MapPin className="w-3 h-3 shrink-0" /> <span className="truncate">{card.college}, {card.location}</span>
                        </button>
                      </div>
                    </div>
                  </div>

                  <div className="flex-1 space-y-4 mb-6">
                    <div>
                      <p className="text-[13px] text-[#6B7280] mb-2 font-medium">Teaching</p>
                      <div className="flex flex-wrap gap-2">
                        {card.teaching.map(s => (
                          <button key={s} onClick={(e) => handleSkillClick(e, s, 'teaching')} className="inline-flex items-center gap-1 px-2.5 py-1 rounded-md bg-[#22C55E]/10 text-[#22C55E] text-[13px] font-medium border border-[#22C55E]/20 hover:bg-[#22C55E]/20 transition-colors">
                            <CheckCircle2 className="w-3.5 h-3.5" /> {s}
                          </button>
                        ))}
                      </div>
                    </div>
                    <div>
                      <p className="text-[13px] text-[#6B7280] mb-2 font-medium">Learning</p>
                      <div className="flex flex-wrap gap-2">
                        {card.wants.map(s => (
                          <button key={s} onClick={(e) => handleSkillClick(e, s, 'learning')} className="inline-flex items-center gap-1 px-2.5 py-1 rounded-md bg-[#7C3AED]/10 text-[#7C3AED] text-[13px] font-medium border border-[#7C3AED]/20 hover:bg-[#7C3AED]/20 transition-colors">
                            <CheckCircle2 className="w-3.5 h-3.5" /> {s}
                          </button>
                        ))}
                      </div>
                    </div>
                  </div>

                  <div className="flex flex-col gap-2 mt-auto pt-4 border-t border-slate-100">
                    <div className="flex items-center justify-between">
                      <button className="text-[13px] text-[#6B7280] hover:text-[#5B5BFF] transition-colors" onClick={(e) => { e.stopPropagation(); handleSwapsClick(e); }}>
                        <span className="font-bold text-[#111827] group-hover:text-[#5B5BFF]">{card.swaps}</span> swaps
                      </button>
                      <button onClick={(e) => { e.stopPropagation(); navigate(`/profile/${card.id}`); }} className="px-4 py-2 rounded-lg bg-slate-100 text-slate-700 text-[13px] font-semibold hover:bg-slate-200 transition-all">
                        View Profile
                      </button>
                    </div>
                    <button onClick={(e) => { e.stopPropagation(); navigate(`/messages/${card.id}`); toast.success(`Started a chat with ${card.name} to learn!`); }} className="w-full px-5 py-2.5 rounded-xl bg-gradient-to-r from-[#5B5BFF] to-[#7C3AED] text-white text-[14px] font-semibold shadow-md hover:shadow-lg hover:brightness-110 hover:scale-[1.02] active:scale-95 transition-all flex items-center justify-center gap-2">
                      <MessageSquare className="w-4 h-4" /> Learn Skill (Text Person)
                    </button>
                  </div>
                </motion.div>
              ))
            ) : (
              <div className="col-span-full py-20 flex flex-col items-center justify-center text-center">
                <div className="w-20 h-20 bg-slate-100 rounded-full flex items-center justify-center mb-4">
                  <Search className="w-10 h-10 text-slate-400" />
                </div>
                <h3 className="text-xl font-bold mb-2">No mentors found.</h3>
                <p className="text-[#6B7280] mb-6">Try changing your filters or search query.</p>
                <button onClick={handleClearFilters} className="px-6 py-3 bg-white border border-slate-200 rounded-xl font-medium hover:bg-slate-50 transition-colors shadow-sm">
                  Clear All Filters
                </button>
              </div>
            )}
          </div>
        </div>
      </main>

      {/* Right Sidebar Filters */}
      <aside className={`fixed inset-y-0 right-0 z-40 w-[320px] bg-white border-l border-slate-200 transform transition-transform duration-300 ease-in-out xl:translate-x-0 xl:static xl:flex xl:flex-col ${isFilterOpen ? 'translate-x-0' : 'translate-x-full'}`}>
        <div className="flex items-center justify-between p-6 border-b border-slate-100">
          <h2 className="text-[22px] font-semibold">Filters</h2>
          <button className="xl:hidden p-2 -mr-2 text-slate-500 hover:text-slate-900 transition-colors" onClick={() => setIsFilterOpen(false)}><X className="w-6 h-6" /></button>
        </div>
        
        <div className="flex-1 overflow-y-auto p-6 space-y-8 hide-scrollbar">
          {/* Level */}
          <div>
            <h3 className="text-[15px] font-semibold mb-4">Level</h3>
            <div className="space-y-3">
              {['Beginner', 'Intermediate', 'Advanced'].map((level) => (
                <button key={level} onClick={() => togglePendingLevel(level)} className="w-full flex items-center gap-3 cursor-pointer group text-left">
                  <div className={`w-5 h-5 rounded-[6px] border flex items-center justify-center transition-colors ${pendingLevels.includes(level) ? 'bg-[#5B5BFF] border-[#5B5BFF]' : 'border-slate-300 group-hover:border-[#5B5BFF]'}`}>
                    {pendingLevels.includes(level) && <CheckCircle2 className="w-4 h-4 text-white" />}
                  </div>
                  <span className={`text-[15px] ${pendingLevels.includes(level) ? 'text-[#111827] font-medium' : 'text-[#6B7280] group-hover:text-[#111827] transition-colors'}`}>{level}</span>
                </button>
              ))}
            </div>
          </div>
          
          {/* Mode */}
          <div>
            <h3 className="text-[15px] font-semibold mb-4">Mode</h3>
            <div className="space-y-3">
              {['Online', 'Offline', 'Hybrid'].map((mode) => (
                <button key={mode} onClick={() => setPendingMode(prev => prev === mode ? null : mode)} className="w-full flex items-center gap-3 cursor-pointer group text-left">
                  <div className={`w-5 h-5 rounded-full border flex items-center justify-center transition-colors ${pendingMode === mode ? 'bg-white border-[#5B5BFF] border-[6px]' : 'border-slate-300 group-hover:border-[#5B5BFF]'}`}>
                  </div>
                  <span className={`text-[15px] ${pendingMode === mode ? 'text-[#111827] font-medium' : 'text-[#6B7280] group-hover:text-[#111827] transition-colors'}`}>{mode}</span>
                </button>
              ))}
            </div>
          </div>

          {/* Availability */}
          <div>
            <h3 className="text-[15px] font-semibold mb-4">Availability</h3>
            <div className="space-y-3">
              {['Weekdays', 'Weekends', 'Anytime'].map((avail) => (
                <button key={avail} onClick={() => togglePendingAvailability(avail)} className="w-full flex items-center gap-3 cursor-pointer group text-left">
                  <div className={`w-5 h-5 rounded-[6px] border flex items-center justify-center transition-colors ${pendingAvailability.includes(avail) ? 'bg-[#5B5BFF] border-[#5B5BFF]' : 'border-slate-300 group-hover:border-[#5B5BFF]'}`}>
                    {pendingAvailability.includes(avail) && <CheckCircle2 className="w-4 h-4 text-white" />}
                  </div>
                  <span className={`text-[15px] ${pendingAvailability.includes(avail) ? 'text-[#111827] font-medium' : 'text-[#6B7280] group-hover:text-[#111827] transition-colors'}`}>{avail}</span>
                </button>
              ))}
            </div>
          </div>
          
          {/* Rating */}
          <div>
            <div className="flex items-center justify-between mb-4">
              <h3 className="text-[15px] font-semibold">Minimum Rating</h3>
              {pendingMinRating > 0 && <span className="text-[13px] text-[#5B5BFF] font-medium">{pendingMinRating}+ Stars</span>}
            </div>
            <div className="flex items-center gap-2">
              {[1,2,3,4,5].map(star => (
                <button 
                  key={star} 
                  onClick={() => setPendingMinRating(star === pendingMinRating ? 0 : star)}
                  className={`p-1 transition-all hover:scale-110 ${star <= pendingMinRating ? 'text-[#F59E0B]' : 'text-slate-200 hover:text-[#F59E0B]/50'}`}
                >
                  <Star className={`w-8 h-8 ${star <= pendingMinRating ? 'fill-current' : ''}`} />
                </button>
              ))}
            </div>
          </div>
        </div>

        <div className="p-6 border-t border-slate-100 bg-white space-y-3">
          <button onClick={handleApplyFilters} className="w-full py-3.5 rounded-xl bg-gradient-to-r from-[#5B5BFF] to-[#7C3AED] text-white text-[15px] font-semibold shadow-lg shadow-[#5B5BFF]/30 hover:brightness-110 hover:scale-[1.02] active:scale-[0.98] transition-all">
            Apply Filters
          </button>
          <button onClick={handleClearFilters} className="w-full py-3.5 rounded-xl bg-slate-50 text-[#6B7280] hover:text-[#111827] hover:bg-slate-100 text-[15px] font-medium transition-all">
            Clear Filters
          </button>
        </div>
      </aside>

      {/* Mobile Overlays */}
      {(isSidebarOpen || isFilterOpen) && (
        <motion.div 
          initial={{ opacity: 0 }}
          animate={{ opacity: 1 }}
          exit={{ opacity: 0 }}
          className="fixed inset-0 bg-[#111827]/40 backdrop-blur-sm z-30 lg:hidden xl:hidden"
          onClick={() => { setIsSidebarOpen(false); setIsFilterOpen(false); }}
        />
      )}

      {/* Logout Confirmation Modal */}
      <AnimatePresence>
        {isLogoutModalOpen && (
          <div className="fixed inset-0 z-50 flex items-center justify-center p-4">
            <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }} className="absolute inset-0 bg-[#111827]/40 backdrop-blur-sm" onClick={() => setIsLogoutModalOpen(false)} />
            <motion.div initial={{ opacity: 0, scale: 0.95, y: 20 }} animate={{ opacity: 1, scale: 1, y: 0 }} exit={{ opacity: 0, scale: 0.95, y: 20 }} className="relative bg-white rounded-2xl p-6 w-full max-w-sm shadow-xl">
              <h3 className="text-xl font-bold mb-2">Logout</h3>
              <p className="text-[#6B7280] mb-6">Are you sure you want to log out of your account?</p>
              <div className="flex gap-3">
                <button onClick={() => setIsLogoutModalOpen(false)} className="flex-1 py-3 rounded-xl bg-slate-50 text-[#111827] font-medium hover:bg-slate-100 transition-colors">Cancel</button>
                <button onClick={handleLogout} className="flex-1 py-3 rounded-xl bg-red-500 text-white font-medium hover:bg-red-600 shadow-md shadow-red-500/20 transition-all">Logout</button>
              </div>
            </motion.div>
          </div>
        )}
      </AnimatePresence>

      {/* All Skills Modal */}
      <AnimatePresence>
        {isSkillsModalOpen && (
          <div className="fixed inset-0 z-50 flex items-center justify-center p-4">
            <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }} className="absolute inset-0 bg-[#111827]/40 backdrop-blur-sm" onClick={() => setIsSkillsModalOpen(false)} />
            <motion.div initial={{ opacity: 0, scale: 0.95, y: 20 }} animate={{ opacity: 1, scale: 1, y: 0 }} exit={{ opacity: 0, scale: 0.95, y: 20 }} className="relative bg-white rounded-2xl p-6 w-full max-w-lg shadow-xl max-h-[80vh] flex flex-col">
              <div className="flex justify-between items-center mb-6">
                <h3 className="text-xl font-bold">All Available Skills ({allAvailableSkills.length})</h3>
                <button onClick={() => setIsSkillsModalOpen(false)} className="p-2 text-slate-400 hover:text-slate-900 transition-colors"><X className="w-5 h-5" /></button>
              </div>
              <div className="flex-1 overflow-y-auto hide-scrollbar">
                <div className="flex flex-wrap gap-2">
                  {allAvailableSkills.map(skill => (
                    <button 
                      key={skill} 
                      onClick={(e) => {
                        setIsSkillsModalOpen(false);
                        handleSkillClick(e, skill, 'teaching');
                      }}
                      className="px-4 py-2 rounded-xl bg-slate-50 border border-slate-200 text-[14px] font-medium text-slate-700 hover:bg-[#5B5BFF] hover:text-white hover:border-[#5B5BFF] transition-colors"
                    >
                      {skill}
                    </button>
                  ))}
                </div>
              </div>
            </motion.div>
          </div>
        )}
      </AnimatePresence>

      <style dangerouslySetInnerHTML={{__html: `
        .hide-scrollbar::-webkit-scrollbar {
          display: none;
        }
        .hide-scrollbar {
          -ms-overflow-style: none;
          scrollbar-width: none;
        }
      `}} />
    </motion.div>
  );
}
